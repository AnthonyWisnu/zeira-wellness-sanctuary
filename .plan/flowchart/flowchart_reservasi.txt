flowchart TD
    %% Node Utama
    START(["Mulai: Pelanggan Memilih Sesi Kelas"])
    
    CHECK_WINDOW{"Jadwal Sesuai Vision Window?<br/>(Member <= 30 Hari, Tamu <= 7 Hari)"}
    REJECT_WINDOW["Pemesanan Ditolak:<br/>Jadwal Belum Terbuka"]
    
    LOCK_CHECK["Kunci Antrean & Periksa Kuota Kursi<br/>PostgreSQL SELECT ... FOR UPDATE"]
    CHECK_SEAT{"Sisa Kursi Studio Masih Tersedia?"}
    REJECT_FULL["Pemesanan Ditolak:<br/>Kuota Kursi Studio Penuh"]
    
    HOLD_PAY["Tahan Kursi Sementara & Buka Pembayaran Midtrans<br/>• Tetapkan Tarif (Member vs Tamu)<br/>• Tahan Kursi (+1) Berstatus Menunggu Pembayaran (15 Menit)<br/>• Terbitkan Modal Pembayaran Snap (QRIS / Virtual Account)"]
    
    CHECK_PAY{"Pembayaran Selesai dalam 15 Menit?<br/>(Verifikasi Webhook & Segel Kriptografis SHA-512)"}
    
    SUCCESS_TICKET["Terbitkan Tiket Resmi (Dipesan)<br/>• Status Pesanan: Lunas<br/>• E-Ticket Sah Digunakan untuk Masuk Kelas"]
    RELEASE_SEAT["Pelepasan Kursi Otomatis (-1)<br/>• Status Pesanan: Kedaluwarsa / Dibatalkan<br/>• Kursi Studio Otomatis Kembali Terbuka"]
    
    END_SUCCESS(["Selesai: Tiket Aktif"])
    END_FAIL(["Selesai: Kursi Bebas"])

    %% Alur Penghubung
    START --> CHECK_WINDOW
    CHECK_WINDOW -->|Ya| LOCK_CHECK
    CHECK_WINDOW -->|Tidak| REJECT_WINDOW
    
    LOCK_CHECK --> CHECK_SEAT
    CHECK_SEAT -->|Tersedia| HOLD_PAY
    CHECK_SEAT -->|Penuh| REJECT_FULL
    
    HOLD_PAY --> CHECK_PAY
    CHECK_PAY -->|Lunas / Berhasil| SUCCESS_TICKET
    CHECK_PAY -->|Batal / Waktu 15 Menit Habis| RELEASE_SEAT
    
    SUCCESS_TICKET --> END_SUCCESS
    RELEASE_SEAT --> END_FAIL
    REJECT_WINDOW --> END_FAIL
    REJECT_FULL --> END_FAIL

    %% Definisi Gaya Visual Elegan
    classDef terminal fill:#0f172a,stroke:#020617,stroke-width:2px,color:#ffffff,font-weight:bold
    classDef process fill:#f8fafc,stroke:#475569,stroke-width:1.5px,color:#0f172a
    classDef decision fill:#eff6ff,stroke:#2563eb,stroke-width:1.5px,color:#1e3a8a,font-weight:bold
    classDef hold fill:#fffbeb,stroke:#d97706,stroke-width:1.5px,color:#92400e
    classDef success fill:#ecfdf5,stroke:#059669,stroke-width:2px,color:#065f46,font-weight:bold
    classDef fail fill:#fef2f2,stroke:#dc2626,stroke-width:1.5px,color:#991b1b

    %% Penerapan Gaya
    class START,END_SUCCESS,END_FAIL terminal
    class LOCK_CHECK process
    class CHECK_WINDOW,CHECK_SEAT,CHECK_PAY decision
    class HOLD_PAY hold
    class SUCCESS_TICKET success
    class REJECT_WINDOW,REJECT_FULL,RELEASE_SEAT fail
